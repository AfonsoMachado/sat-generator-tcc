ALGORITMO Gerar_Formulas_KSAT(n, num_lits, min_clauses, max_clausulas, min_lits, max_lits, k)
    INICIO
        literais <- ["X1" ... "Xnum_lits"]
        valoracao <- {lit : sorteio(V/F) PARA cada lit em literais}

        PARA i DE 1 ATÉ n FAÇA
            num_clausulas ← sorteio(min_clauses, max_clausulas)
            max_lits_formula ← sorteio(min_lits, max_lits)

            lits_escolhidos ← sorteio_subconjunto(literais, max_lits_formula)

            formula <- []

            PARA c DE 1 ATÉ num_clausulas FAÇA
                clausula_lits ← sorteio_subconjunto(lits_escolhidos, minimo(k, tamanho(lits_escolhidos)))
                clausula <- []

                PARA cada lit em clausula_lits FAÇA
                    SE sorteio(V/F) = VERDADEIRO
                        clausula.adicionar(lit)
                    SENAO
                        clausula.adicionar("¬" + lit)
                    FIM SE
                FIM PARA

                formula.adicionar(clausula)
            FIM PARA

            fnc <- concatenar([ "(" + concatenar(clausula, " ∨ ") + ")" PARA cada clausula em formula ], " ∧ ")
            ESCREVA("Fórmula", i)
            ESCREVA("FNC:", conjuncao_disjuncao(formula))
            PARA cada lit em lits_escolhidos FAÇA
                ESCREVA(lit, "=", valoracao[lit])
            FIM PARA
                        
            // --- Avaliação da fórmula ---
            valor_formula ← VERDADEIRO
            PARA cada cláusula EM fórmula FAÇA
                valor_clausula ← FALSO
                PARA cada literal EM cláusula FAÇA
                    SE literal COMEÇA_COM "¬" ENTÃO
                        var ← remover_prefixo("¬", literal)
                        valor_literal ← NÃO valoração[var]
                    SENÃO
                        var ← literal
                        valor_literal ← valoração[var]
                    FIM SE
                    valor_clausula ← valor_clausula OU valor_literal
                FIM PARA
                valor_formula ← valor_formula E valor_clausula
            FIM PARA

            ESCREVA("Valoração da fórmula:", valor_formula)
        FIM PARA
    FIM
