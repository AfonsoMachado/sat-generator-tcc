ALGORITMO Gerar_Formulas_KSAT(n, num_lits, min_clauses, max_clausulas, min_lits, max_lits, k)
    INICIO
        literais <- ["X1" ... "Xnum_lits"]
        valoracao <- {lit : sorteio(V/F) PARA cada lit em literais}

        PARA i DE 1 ATÉ n FAÇA
            num_clausulas ← sorteio(min_clauses, max_clausulas)
            max_lits_formula ← sorteio(min_lits, max_lits)

            lits_escolhidos ← sorteio_subconjunto(literais, max_lits_formula)

            formula <- []

            PARA c DE 1 ATÉ num_clausulas FAÇA
                clausula_lits ← sorteio_subconjunto(lits_escolhidos, minimo(k, tamanho(lits_escolhidos)))
                clausula <- []

                PARA cada var em clausula_lits FAÇA
                    SE sorteio(V/F) = VERDADEIRO
                        clausula.adicionar(var)
                    SENAO
                        clausula.adicionar("¬" + var)
                    FIM SE
                FIM PARA

                formula.adicionar(clausula)
            FIM PARA

            fnc <- juntar([ "(" + juntar(clausula, " ∨ ") + ")" PARA cada clausula em formula ], " ∧ ")
            ESCREVA("Fórmula", i)
            ESCREVA("FNC:", conjuncao_disjuncao(formula))
            PARA cada var em lits_escolhidos FAÇA
                ESCREVA(var, "=", valoracao[var])
            FIM PARA
        FIM PARA
    FIM
