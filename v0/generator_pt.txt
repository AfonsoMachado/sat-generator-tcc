ALGORITMO Gerar_Formulas_KSAT(n, num_vars, min_clauses, max_clausulas, min_vars, max_vars, k)
    INICIO
        literais <- ["X1" ... "Xnum_vars"]
        valoracao <- {lit : sorteio(V/F) PARA cada lit em literais}

        PARA i DE 1 ATÉ n FAÇA
            num_clausulas ← sorteio(min_clauses, max_clausulas)
            max_vars_formula ← sorteio(min_vars, max_vars)

            vars_escolhidas ← sorteio_subconjunto(literais, max_vars_formula)

            formula <- []

            PARA c DE 1 ATÉ num_clausulas FAÇA
                clausula_vars ← sorteio_subconjunto(vars_escolhidas, minimo(k, tamanho(vars_escolhidas)))
                clausula <- []

                PARA cada var em clausula_vars FAÇA
                    SE sorteio(V/F) = VERDADEIRO
                        clausula.adicionar(var)
                    SENAO
                        clausula.adicionar("¬" + var)
                    FIM SE
                FIM PARA

                formula.adicionar(clausula)
            FIM PARA

            fnc <- juntar([ "(" + juntar(clausula, " ∨ ") + ")" PARA cada clausula em formula ], " ∧ ")
            ESCREVA("Fórmula", i)
            ESCREVA("FNC:", conjuncao_disjuncao(formula))
            PARA cada var em vars_escolhidas FAÇA
                ESCREVA(var, "=", valoracao[var])
            FIM PARA
        FIM PARA
    FIM
